# Seed modules only.
# Keep this list short: transitive dependencies are resolved automatically.
# Prefix boot-critical seeds with 'base:' to include them in early-load list.
# Use 'normal:' (or no prefix) for modules loaded after boot UI appears.
# Use 'auto:' for drivers that should load only when matching hardware exists
# (by modalias, at boot and on hotplug). auto seeds may be globs.
# Add a board filter to limit a seed to some boards (pi4, pi5, qemu):
#   base(pi5): drivers/foo        only on Pi 5
#   normal(pi4,pi5): drivers/bar  on Pi 4 and Pi 5
#   normal(!pi5): drivers/baz     on every board except Pi 5

# Pi 5 fan header cooling. PWM providers are not symbol dependencies of
# pwm-fan; seed RP1 explicitly (the supplied Pi kernels build it in).
base(pi5): drivers/pwm/pwm-rp1
base(pi5): drivers/hwmon/pwm-fan

# Pi 5 onboard power button reports KEY_POWER through gpio-keys (evdev is built in).
base(pi5): drivers/input/keyboard/gpio_keys

# Display and touch stack. These are hardware-loaded ("auto"): every module is
# staged with its dependencies, and init loads only the drivers whose modalias
# matches a device that exists (coldplug at boot, kernel uevents afterwards).
# This covers any KMS display (DSI panels, HDMI/mini-HDMI, DPI, SPI TFTs, USB
# displays), their bridges/regulators/backlights, and touch controllers on I2C,
# SPI or USB, without per-display seed lists. Full KMS on Pi 4 needs the
# brcmstb DDC I2C controller for HDMI or vc4 never binds (including DSI).
auto(!qemu): drivers/gpu/drm/*
auto(!qemu): drivers/video/backlight/*
auto(!qemu): drivers/video/fbdev/*
auto(!qemu): drivers/staging/fbtft/*
auto(!qemu): drivers/regulator/*
auto(!qemu): drivers/i2c/busses/*
auto(!qemu): drivers/i2c/muxes/*
auto(!qemu): drivers/spi/spi-bcm2835*
auto(!qemu): drivers/pwm/*
auto(!qemu): drivers/input/touchscreen/*
auto(!qemu): drivers/hid/hid-multitouch
auto(!qemu): drivers/hid/i2c-hid/*
auto(!qemu): drivers/hid/wacom
auto(!qemu): sound/soc/codecs/snd-soc-hdmi-codec
# QEMU debug build (make debug) uses a virtio-gpu display instead.
base(qemu): drivers/gpu/drm/virtio/virtio-gpu
base: drivers/i2c/i2c-dev

# The network stack for providing the ethernet gadget for the web ui
normal: drivers/usb/gadget/function/u_ether
normal: drivers/usb/gadget/function/usb_f_ncm

# QEMU debug build (make debug): virtio NIC for USB/IP and port forwards, and
# the virtual USB device controller that exports the gadget over USB/IP.
normal(qemu): drivers/net/virtio_net
normal(qemu): drivers/usb/usbip/usbip-vudc
